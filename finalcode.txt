from ultralytics import YOLO
import cv2, pandas as pd, os, random, time

print("Current Working Directory:", os.getcwd())

# ----------------------- FIREBASE SETUP (for GPS) -----------------------
service_account_path = r"E:\restart\pothole-detection-71587-firebase-adminsdk-fbsvc-34543a18de.json"
firebase_initialized = False
try:
    import firebase_admin
    from firebase_admin import credentials, db
    cred = credentials.Certificate(service_account_path)
    firebase_admin.initialize_app(cred, {
        'databaseURL': 'https://pothole-detection-71587-default-rtdb.firebaseio.com'
    })
    firebase_initialized = True
    print("Initialized Firebase Realtime DB connection.")
except Exception as e:
    print("Firebase init failed or firebase_admin not installed:", e)
    print("Falling back to dummy GPS.")

# ----------------------- GPS FUNCTION -----------------------
def get_gps():
    base_lat = 9.5916
    base_lon = 76.5222
    if firebase_initialized:
        try:
            ref = db.reference('location')
            data = ref.get()
            if data and 'latitude' in data and 'longitude' in data:
                lat = float(data.get('latitude', base_lat))
                lon = float(data.get('longitude', base_lon))
                return lat, lon
            else:
                print("No valid 'location' data in Firebase; using dummy GPS.")
        except Exception as e:
            print("Error reading location from Firebase:", e)
    return base_lat + random.uniform(-0.0005, 0.0005), base_lon + random.uniform(-0.0005, 0.0005)

# ----------------------- LOAD YOLO -----------------------
model = YOLO("best.pt")
names = model.names
 
# ----------------------- CREATE RESULTS FOLDER -----------------------
results_dir = r"E:\Btech\pothole project\pothole_results"
os.makedirs(results_dir, exist_ok=True)
crop_dir = os.path.join(results_dir, "pothole_crops")
os.makedirs(crop_dir, exist_ok=True) 

log = []
sl_no = 1

# ----------------------- TRY TO OPEN CAMERA -----------------------
# ----------------------- CONNECT TO ESP32 CAMERA -----------------------
ESP32_STREAM_URL =  "http://192.168.43.35:81/streamq"  # Replace with your ESP32 IP
cap = cv2.VideoCapture(ESP32_STREAM_URL)
if not cap.isOpened():
    print("Cannot connect to ESP32-CAM stream")
    print("Warning: Camera not accessible. Using dummy images/data instead.")
    # Create dummy table with images
    for i in range(1, 6):
        crop_name = f"Pothole_{i}.jpg"
        crop_path = os.path.join(crop_dir, crop_name)
        with open(crop_path, "wb") as f:
            f.write(b"")  # empty image
        # use get_gps() even for dummy table to propagate firebase location if available
        lat, lon = get_gps()
        log.append({
            "Sl.No": sl_no,
            "Class": "Pothole",
            "Width(px)": random.randint(20,100),
            "Height(px)": random.randint(20,100),
            "Latitude": lat,
            "Longitude": lon,
            "Image File": crop_path
        })
        sl_no += 1
else:
    print("Connected to ESP32-CAM successfully!")
    print("Camera opened successfully. Press 'q' to quit.")


    frame_count = 0
    stream_failures = 0
    while True:
        ret, frame = cap.read()
        if not ret:
            stream_failures += 1
            print(f"Warning: Failed to read frame from camera (attempt {stream_failures})")
            if stream_failures >= 3:
                print("Too many failed frame reads, exiting loop.")
                break
            time.sleep(0.5)
            continue
        # successful read
        stream_failures = 0
        frame_count += 1
        frame = cv2.resize(frame, (640, 480))
        lat, lon = get_gps()
        # debug: show where GPS is coming from each frame
        print(f"gps -> {lat:.6f}, {lon:.6f}")

        results = model(frame, verbose=False)[0]
        for box in results.boxes:
            conf = float(box.conf)
            if conf < 0.1:  # low threshold
                continue
            cls = int(box.cls)
            label = names[cls]
            x1, y1, x2, y2 = map(int, box.xyxy[0])
            w, h = x2 - x1, y2 - y1
            crop_name = f"{label}_f{frame_count}.jpg"
            crop_path = os.path.join(crop_dir, crop_name)
            cv2.imwrite(crop_path, frame[y1:y2, x1:x2])
            cv2.rectangle(frame, (x1, y1), (x2, y2), (0,0,255), 2)
            cv2.putText(frame, f"{label} {conf:.2f}", (x1, y1-5),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255,255,255), 2)
            log.append({
                "Sl.No": sl_no,
                "Class": label,
                "Width(px)": w,
                "Height(px)": h,
                "Latitude": lat,
                "Longitude": lon,
                "Image File": crop_path
            })
            sl_no += 1
        cv2.imshow("YOLO Detection", frame)
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
    cap.release()
    cv2.destroyAllWindows()

# ----------------------- ENSURE TABLE -----------------------
if len(log) == 0:
    log.append({
        "Sl.No": 1,
        "Class": "None",
        "Width(px)": 0,
        "Height(px)": 0,
        "Latitude": 0,
        "Longitude": 0,
        "Image File": "None"
    })

df = pd.DataFrame(log)
csv_file = os.path.join(results_dir, "pothole_final.csv")
excel_file = os.path.join(results_dir, "pothole_final.xlsx")
df.to_csv(csv_file, index=False)
df.to_excel(excel_file, index=False)

print("\nResults saved successfully!")
print(f"Folder: {results_dir}")
print(f"CSV: {csv_file}")
print(f"Excel: {excel_file}")
print(f"Cropped images: {crop_dir}")
print("\nTable preview:")
print(df.head())
